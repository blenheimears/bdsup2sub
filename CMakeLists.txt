cmake_minimum_required(VERSION 3.16)
project(bdsup2sub_native VERSION 0.1.0 LANGUAGES CXX)
find_package(PNG REQUIRED)
add_executable(bdsup2sub src/main.cpp)
target_compile_features(bdsup2sub PRIVATE cxx_std_17)
target_link_libraries(bdsup2sub PRIVATE PNG::PNG)
if(MSVC)
  target_compile_options(bdsup2sub PRIVATE /W4)
else()
  target_compile_options(bdsup2sub PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS bdsup2sub RUNTIME DESTINATION bin)
option(BDSUP2SUB_NATIVE_BUILD_TESTS "Enable native subtitle converter tests" ${PROJECT_IS_TOP_LEVEL})
if(BDSUP2SUB_NATIVE_BUILD_TESTS)
  include(CTest)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME bdsup2sub-fixtures COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixture_test.py $<TARGET_FILE:bdsup2sub>)
  add_test(NAME bdsup2sub-help COMMAND bdsup2sub --help)
  add_test(NAME bdsup2sub-version COMMAND bdsup2sub --version)
endif()
